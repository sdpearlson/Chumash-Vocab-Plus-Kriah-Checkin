<script>


// --- YOUR APPS SCRIPT URL ---


// Paste the URL you got in Step 2 here!


const GAS_WEB_APP_URL = "https://script.google.com/macros/s/AKfycbxOsTcNWamk4Sk2DM6175X7uHjWM3ICSesbcUWJYWmxogqnbNJpKHp3oStb-1ch-VGu/exec";


 


const state = { name: "", selfEval: null, history: [], currentRank: 1, questionCount: 0 };


let mediaRecorder;


let audioChunks = [];


let isRecording = false;


let recordingTimer = null;


 


// Gemini's inline-audio request cap is 20MB (prompt + base64 audio combined).


// Auto-stopping recordings keeps every submission comfortably under that limit


// instead of silently failing on longer readings.


const MAX_RECORDING_SECONDS = 90;


// Safety check on the raw blob before upload; base64 inflates size ~33%,


// so keep raw audio well under the 20MB post-encoding ceiling.


const MAX_BLOB_BYTES = 12 * 1024 * 1024; // ~12MB raw


 


document.getElementById('nameSubmitBtn').addEventListener('click', () => {


    state.name = document.getElementById('studentName').value.trim();


    if (!state.name) return alert("Please enter your name.");


    document.getElementById('displayName').innerText = state.name;


    hide('nameArea'); show('hubMenuArea');


});


 


function hide(id) { document.getElementById(id).classList.add('hidden'); }


function show(id) { document.getElementById(id).classList.remove('hidden'); }


 


// (Paste your FULL 41-word QUESTION_BANK array right here)


const TOTAL_QUESTIONS = 10;


const QUESTION_BANK = [


  { ri: 1, rank: 1, display: "אֶרֶץ", meaning: "land", options: ["land", "sky", "water", "light"] },


  { ri: 11, rank: 2, display: "אֲשֶׁר", meaning: "that / which", options: ["that / which", "when", "there", "because"] },


  { ri: 25, rank: 3, display: "פֶּן", meaning: "lest / so that not", options: ["lest / so that not", "maybe", "also", "here"] }


];


 


function startVocab() { hide('hubMenuArea'); show('selfEvalArea'); }


function setVocabLevel(evalName, rank) {


    state.selfEval = evalName; state.currentRank = rank;


    hide('selfEvalArea'); show('quizArea'); show('quizProgress'); runNext();


}


 


function runNext() {


    state.questionCount++;


    if (state.questionCount > TOTAL_QUESTIONS) return finishVocab();


    document.getElementById('progressBar').style.width = ((state.questionCount-1)/TOTAL_QUESTIONS*100)+'%';


    let potential = QUESTION_BANK.filter(q => q.rank === state.currentRank && !state.history.find(h => h.ri === q.ri));


    if (potential.length === 0) potential = QUESTION_BANK.filter(q => !state.history.find(h => h.ri === q.ri));


    const q = potential[Math.floor(Math.random()*potential.length)];


    const choices = [...q.options].sort(() => Math.random() - 0.5);


    document.getElementById('quizArea').innerHTML = `


      <div class="intake-hebrew-word">${q.display}</div>


      <div class="intake-choices">${choices.map(o => `<button class="intake-choice" onclick="window.ans('${o}', '${q.meaning}', ${q.ri}, '${q.display}', ${q.rank})">${o}</button>`).join('')}</div>


      <button class="intake-skip" onclick="window.ans(null, '${q.meaning}', ${q.ri}, '${q.display}', ${q.rank})">I'm not sure &rarr;</button>`;


}


 


window.ans = (picked, correct, ri, word, rank) => {


    state.history.push({ ri, word, correct: (picked === correct), picked: picked ? picked : "Skipped" });


    if((picked === correct) && state.currentRank < 3) state.currentRank++;


    else if(!(picked === correct) && state.currentRank > 1) state.currentRank--;


    runNext();


};


 


function finishVocab() {


    hide('quizArea'); hide('quizProgress');


    document.getElementById('resultsDesc').innerHTML = `Checking answers...`; show('resultsArea');


    const score = state.history.filter(h => h.correct).length;


 


    // FETCH TO EXTERNAL APPS SCRIPT URL


    fetch(GAS_WEB_APP_URL, {


        method: "POST",


        body: JSON.stringify({


            action: "saveVocab",


            payload: { name: state.name, selfEval: state.selfEval, finalRank: state.currentRank, score: score, total: TOTAL_QUESTIONS, history: state.history }


        }),


        headers: { "Content-Type": "text/plain;charset=utf-8" }


    })


    .then(res => res.json())


    .then(data => {


        if (data.success) {


            document.getElementById('resultsDesc').innerHTML = `Vocab Check-In logged successfully!`;


        } else {


            document.getElementById('resultsDesc').innerHTML = `Error saving data: ${data.error || 'Unknown error'}`;


        }


    })


    .catch(err => { document.getElementById('resultsDesc').innerHTML = `Error saving data: ${err}`; });


}


 


function startAudio() { hide('hubMenuArea'); show('audioArea'); }


 


async function toggleRecording() {


    const btn = document.getElementById('recordBtn');


    const status = document.getElementById('audioStatus');


 


    if (!isRecording) {


        try {


            const stream = await navigator.mediaDevices.getUserMedia({ audio: true });


            mediaRecorder = new MediaRecorder(stream);


            audioChunks = [];


            mediaRecorder.ondataavailable = e => audioChunks.push(e.data);


            mediaRecorder.onstop = sendAudioToServer;


            mediaRecorder.start();


            isRecording = true;


            btn.innerText = "⏹ Stop & Submit";


            btn.classList.add("recording");


 


            // Auto-stop at MAX_RECORDING_SECONDS so uploads stay under Gemini's


            // inline-audio request limit. Shows a live countdown too.


            let secondsLeft = MAX_RECORDING_SECONDS;


            status.innerText = `Recording... Read clearly. Auto-stops in ${secondsLeft}s.`;


            recordingTimer = setInterval(() => {


                secondsLeft--;


                if (secondsLeft <= 0) {


                    clearInterval(recordingTimer);


                    if (isRecording) toggleRecording();


                } else {


                    status.innerText = `Recording... Read clearly. Auto-stops in ${secondsLeft}s.`;


                }


            }, 1000);


        } catch (err) {


            alert("Microphone access denied. Please check your browser settings.");


        }


    } else {


        clearInterval(recordingTimer);


        mediaRecorder.stop();


        isRecording = false;


        btn.innerText = "Uploading...";


        btn.classList.remove("recording");


        btn.disabled = true;


        status.innerText = "Analyzing reading with AI... this takes about 10-15 seconds.";


    }


}


 


function sendAudioToServer() {


    const audioBlob = new Blob(audioChunks, { type: 'audio/webm' });


 


    // Guard against oversized recordings before spending time on upload.


    if (audioBlob.size > MAX_BLOB_BYTES) {


        document.getElementById('audioStatus').innerText =


            "That recording is too long to process reliably. Please re-record a shorter passage (under 90 seconds).";


        document.getElementById('recordBtn').disabled = false;


        document.getElementById('recordBtn').innerText = "🎤 Start Recording";


        return;


    }


 


    const reader = new FileReader();


    reader.readAsDataURL(audioBlob);


    reader.onloadend = function() {


        const base64Audio = reader.result.split(',')[1];


 


        // FETCH TO EXTERNAL APPS SCRIPT URL


        fetch(GAS_WEB_APP_URL, {


            method: "POST",


            body: JSON.stringify({


                action: "processAudio",


                payload: { base64Audio: base64Audio, studentName: state.name }


            }),


            headers: { "Content-Type": "text/plain;charset=utf-8" }


        })


        .then(res => res.json())


        .then(data => {


            if (data.success) {


                hide('audioArea');


                document.getElementById('resultsDesc').innerHTML = `Audio processed! Your teacher has the results.`;


                show('resultsArea');


            } else {


                // Previously this branch never ran, so failures looked identical


                // to successes. Now the real server-side error is shown.


                document.getElementById('audioStatus').innerText = `Error: ${data.error || 'Unknown error'}. Please try recording again.`;


                document.getElementById('recordBtn').disabled = false;


                document.getElementById('recordBtn').innerText = "🎤 Start Recording";


            }


        })


        .catch(err => {


            document.getElementById('audioStatus').innerText = "An error occurred while uploading: " + err;


            document.getElementById('recordBtn').disabled = false;


            document.getElementById('recordBtn').innerText = "🎤 Start Recording";


        });


    }


}


</script>
